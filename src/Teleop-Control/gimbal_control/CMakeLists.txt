cmake_minimum_required(VERSION 3.8)
project(gimbal_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
set(dependencies controller_interface hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools geometry_msgs std_msgs)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/gimbal_controller.cpp src/maestro_system.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} ${dependencies})
pluginlib_export_plugin_description_file(controller_interface controller_plugins.xml)
pluginlib_export_plugin_description_file(hardware_interface hardware_plugins.xml)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_gimbal test/test_gimbal.cpp)
  if(controller_interface_VERSION VERSION_GREATER_EQUAL "3.0.0")
    target_compile_definitions(test_gimbal PRIVATE GIMBAL_INIT_WITH_DESCRIPTION)
  endif()
  target_link_libraries(test_gimbal ${PROJECT_NAME} util)
  ament_target_dependencies(test_gimbal ${dependencies})
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(${dependencies})
ament_package()
