#examples of how to run tests

cmake --build /workspaces/rover/build/morseled
rm -f ~/.cache/gstreamer-1.0/registry*.bin
export GST_PLUGIN_PATH=/workspaces/rover/build/morseled
gst-inspect-1.0 morseLED | rg "Filename|draw-roi|start-detection|decoded-text"

python3 /workspaces/rover/src/Cameras/video_streaming/gst-plugins/morseLED/test_morse_led.py \
  --plugin-path /workspaces/rover/build/morseled \
  --message "ABCDEFGHIJKLMNOPQRSTUVWXYZ 0123456789" \
  --sim-wpm 18 --decode-wpm 18 \
  --framerate 90 \
  --lead-in 1.0 \
  --loop --num-buffers 3500 \
  --roi-x 296 --roi-y 216 --roi-width 48 --roi-height 48

python3 /workspaces/rover/src/Cameras/video_streaming/gst-plugins/morseLED/test_morse_led.py \
  --plugin-path /workspaces/rover/build/morseled \
  --message "SOS" \
  --sim-wpm 18 --decode-wpm 18 \
  --framerate 90 \
  --loop --num-buffers 1500 \
  --random-dot \
  --calibrate --calibration-seconds 1.0 \
  --roi-x 0 --roi-y 0 --roi-width 64 --roi-height 64 \
  --draw-roi
